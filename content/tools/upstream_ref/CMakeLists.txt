# The original CATHODE machine, unmodified, built headless (no JUCE, no app layer) for the A/B renders.
#   cmake -S content/tools/upstream_ref -B <scratch>/upstream_build -DCMAKE_BUILD_TYPE=Release -G Ninja && cmake --build <scratch>/upstream_build -j8
cmake_minimum_required(VERSION 3.20)
project(upstream_ref CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
set(UP ${CMAKE_CURRENT_SOURCE_DIR}/../../../ref/upstream)
file(GLOB DSP_SRCS CONFIGURE_DEPENDS "${UP}/sdk/dsp/src/*.cpp")
file(GLOB_RECURSE PLAITS CONFIGURE_DEPENDS "${UP}/sdk/dsp/thirdparty/plaits/*.cc")
file(GLOB CORE_SRCS CONFIGURE_DEPENDS "${UP}/sdk/core/src/*.cpp")
add_library(up_plaits STATIC ${PLAITS})
target_include_directories(up_plaits PUBLIC ${UP}/sdk/dsp/thirdparty/plaits)
target_compile_options(up_plaits PRIVATE -w)
add_library(up_dsp STATIC ${DSP_SRCS})
target_include_directories(up_dsp PUBLIC ${UP}/sdk/dsp/include)
target_link_libraries(up_dsp PUBLIC up_plaits)
target_compile_options(up_dsp PRIVATE -w)
add_library(up_core STATIC ${CORE_SRCS})
target_include_directories(up_core PUBLIC ${UP}/sdk/core/include ${UP}/sdk/machine/include)
target_link_libraries(up_core PUBLIC up_dsp)
target_compile_options(up_core PRIVATE -w)
add_library(up_cathode STATIC ${UP}/cathode/src/CathodeMachine.cpp ${UP}/cathode/src/CathodeFacade.cpp)
target_include_directories(up_cathode PUBLIC ${UP}/cathode/include)
target_link_libraries(up_cathode PUBLIC up_core)
target_compile_options(up_cathode PRIVATE -w)
add_executable(upstream_render upstream_render.cpp)
target_link_libraries(upstream_render PRIVATE up_cathode)
