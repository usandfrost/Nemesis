# Host benchmark of the FX candidates (see fx_bench.cpp). Build OUTSIDE the repo:
#   python3 -I fx_bench/make_tree.py $S/tree
#   cmake -S fx_bench -B $S/build_double -DFX_TREE=$S/tree                                            (Apple clang, Real = double)
#   cmake -S fx_bench -B $S/build_float  -DFX_TREE=$S/tree -DCMAKE_CXX_COMPILER=/opt/homebrew/bin/g++-16 -DFX_FLOAT=ON   (Real = float, firmware flags)
cmake_minimum_required(VERSION 3.20)
project(fx_bench CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
    set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_FLAGS_RELEASE "-O2")
set(FX_TREE "" CACHE PATH "output of make_tree.py")
option(FX_FLOAT "Real = float, fast math (the firmware's DSP configuration)" OFF)
set(FORK_DSP "${CMAKE_CURRENT_SOURCE_DIR}/../firmware/dsp")

file(GLOB FORK_SRCS CONFIGURE_DEPENDS "${FORK_DSP}/src/*.cpp")
file(GLOB_RECURSE FORK_PLAITS CONFIGURE_DEPENDS "${FORK_DSP}/thirdparty/plaits/*.cc")
file(GLOB FX_SRCS CONFIGURE_DEPENDS "${FX_TREE}/src/*.cpp")
set_source_files_properties(${FORK_PLAITS} PROPERTIES COMPILE_OPTIONS -w)

add_library(fxdsp STATIC ${FORK_SRCS} ${FORK_PLAITS} ${FX_SRCS})
target_include_directories(fxdsp PUBLIC "${FX_TREE}/inc" "${FORK_DSP}/include" "${FORK_DSP}/thirdparty/plaits")
target_compile_definitions(fxdsp PUBLIC NEMESIS_FORK_EXACT=1 NEMESIS_FORK_APPROX=1 NEMESIS_FORK_MAX_VOICES=8 DSP_IS_FORK=1)
if(FX_FLOAT)
    target_compile_definitions(fxdsp PUBLIC NEMESIS_DSP_FLOAT=1 NEMESIS_DSP_FAST_MATH=1)
endif()

add_executable(fx_bench fx_bench.cpp)
target_link_libraries(fx_bench PRIVATE fxdsp)
target_include_directories(fx_bench PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
if(FX_FLOAT AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    # as the firmware: unsuffixed literals are float in the converted files. (clang ignores the flag, so float builds use g++.)
    set_source_files_properties(fx_bench.cpp PROPERTIES COMPILE_OPTIONS "-fsingle-precision-constant")
    # the same rule as bench/CMakeLists.txt: every converted source (the ones that include Real.h)
    foreach(_f ${FORK_SRCS} ${FX_SRCS})
        file(READ ${_f} _txt)
        string(FIND "${_txt}" "nemesis/dsp/Real.h" _pos)
        if(NOT _pos EQUAL -1)
            set_source_files_properties(${_f} PROPERTIES COMPILE_OPTIONS "-fsingle-precision-constant")
        endif()
    endforeach()
endif()

add_executable(fx_quality fx_quality.cpp)
target_link_libraries(fx_quality PRIVATE fxdsp)
target_include_directories(fx_quality PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
if(FX_FLOAT AND CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    set_source_files_properties(fx_quality.cpp PROPERTIES COMPILE_OPTIONS "-fsingle-precision-constant")
endif()
